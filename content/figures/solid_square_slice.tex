\draw[->,gray] (0,0,0)--(1.15,0,0)
  node[below right,text=black] {$x$};

  node[above,text=black] {$y$};
\draw[dashed,gray!70] (X0)--(A0) (X1)--(A1);
\coordinate (DX0) at ([xshift=-3pt,yshift=-8pt]X0);
\coordinate (DX1) at ([xshift=-3pt,yshift=-8pt]X1);

  node[midway,below left=3pt] {$dx$};
% The full solid continues to the pinched endpoint at x=1.
\fill[gray!7]
  plot[domain=\cutright:1,samples=35,variable=\t] ({\t},{\t*\t},0)

    ({\t},{\t*\t},{\t-\t*\t})--cycle;

    ({\t},{\t*\t},{\t-\t*\t})--cycle;

  plot[domain=0:\cutleft,samples=45,variable=\t]
