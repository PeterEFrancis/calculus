% Revolution of x^2 <= r <= x, 0 <= x <= 1, with a front cutaway.
% Projection: (x,r,theta) -> (5.7x-.65r cos(theta), 1.65r sin(theta)).
% The washer faces are vertical ellipses; both radii equal 1 at the far rim.
\begin{tikzpicture}[line width=.65pt,line cap=round,line join=round,
  every node/.style={font=\small,inner sep=2pt}]
\pgfmathsetmacro{\washerlowertangent}{-acos(.65/5.7)}
\pgfmathsetmacro{\washerlowerend}{360+\washerlowertangent}

% Draw the gray context continuously beneath the translucent highlight.
% Its lower cut face shares the blue tangent and the single far-rim junction.
\begin{scope}[transparency group,opacity=.7]
\foreach \a/\b in {0/1} {
  \fill[gray!12,variable=\t]
    plot[domain=90:\washerlowerend,samples=81]
      ({5.7*\b-.65*\b*cos(\t)},{1.65*\b*sin(\t)})
    --plot[domain=\washerlowerend:90,samples=81]
      ({5.7*\a-.65*\a*cos(\t)},{1.65*\a*sin(\t)})--cycle;
  \fill[gray!24,variable=\u]
    plot[domain=\a:\b,samples=41] ({5.7*\u},{1.65*\u*\u})
    --plot[variable=\t,domain=90:\washerlowerend,samples=81]
      ({5.7*\b-.65*\b*\b*cos(\t)},{1.65*\b*\b*sin(\t)})
    --plot[variable=\u,domain=\b:\a,samples=41]
      ({5.7*\u-.65*\u*\u*cos(\washerlowertangent)},
       {1.65*\u*\u*sin(\washerlowertangent)})
    --plot[variable=\t,domain=\washerlowerend:90,samples=81]
      ({5.7*\a-.65*\a*\a*cos(\t)},{1.65*\a*\a*sin(\t)})--cycle;
  \fill[gray!10,variable=\u]
    plot[domain=\a:\b,samples=2] ({5.7*\u},{1.65*\u})
    --plot[domain=\b:\a,samples=41] ({5.7*\u},{1.65*\u*\u})--cycle;
  \fill[gray!10,variable=\u]
    plot[domain=\a:\b,samples=2]
      ({5.7*\u-.65*\u*cos(\washerlowertangent)},
       {1.65*\u*sin(\washerlowertangent)})
    --plot[domain=\b:\a,samples=41]
      ({5.7*\u-.65*\u*\u*cos(\washerlowertangent)},
       {1.65*\u*\u*sin(\washerlowertangent)})--cycle;
}
\end{scope}

% Clip the inner-wall reference curves before drawing the stronger borders.
\begin{scope}
\clip[variable=\u]
  plot[domain=0:1,samples=101] ({5.7*\u},{1.65*\u*\u})
  --plot[variable=\t,domain=90:\washerlowerend,samples=101]
    ({5.7-.65*cos(\t)},{1.65*sin(\t)})
  --plot[variable=\u,domain=1:0,samples=101]
    ({5.7*\u-.65*\u*\u*cos(\washerlowertangent)},
     {1.65*\u*\u*sin(\washerlowertangent)})--cycle;
\foreach \q in {.78,.9} {
  \draw[black!30,line width=.4pt,variable=\t,domain=90:\washerlowerend,samples=81]
    plot ({5.7*\q-.65*\q*\q*cos(\t)},{1.65*\q*\q*sin(\t)});
}
\end{scope}
\draw[black!60] (0,0)--(5.7,1.65);
\draw[black!60,variable=\u,domain=0:1,samples=81]
  plot ({5.7*\u},{1.65*\u*\u});
\draw[black!60,variable=\u,domain=0:1,samples=81]
  plot ({5.7*\u-.65*\u*\u*cos(\washerlowertangent)},
        {1.65*\u*\u*sin(\washerlowertangent)});
\draw[black!60] (0,0)--
  ({5.7-.65*cos(\washerlowertangent)},{1.65*sin(\washerlowertangent)});
\draw[black!60,variable=\t,domain=90:\washerlowerend,samples=101]
  plot ({5.7-.65*cos(\t)},{1.65*sin(\t)});

% Composite all blue surfaces once. Complete annular faces and complete
% inner/outer walls share one color, so no front/rear shading seam appears.
\begin{scope}[transparency group,opacity=.22]
\foreach \q in {.57,.63} {
  \fill[figureblue,even odd rule,variable=\t]
    plot[domain=-90:270,samples=121]
      ({5.7*\q-.65*\q*cos(\t)},{1.65*\q*sin(\t)})--cycle
    plot[domain=-90:270,samples=121]
      ({5.7*\q-.65*\q*\q*cos(\t)},{1.65*\q*\q*sin(\t)})--cycle;
}
\foreach \power in {1,2} {
  \pgfmathsetmacro{\ra}{.57^\power}
  \pgfmathsetmacro{\rb}{.63^\power}
  \foreach \t in {0,5,...,355} {
    \pgfmathsetmacro{\nextt}{\t+5}
    \filldraw[fill=figureblue,draw=figureblue,line width=.06pt]
      ({5.7*.57-.65*\ra*cos(\t)},{1.65*\ra*sin(\t)})--
      ({5.7*.63-.65*\rb*cos(\t)},{1.65*\rb*sin(\t)})--
      ({5.7*.63-.65*\rb*cos(\nextt)},{1.65*\rb*sin(\nextt)})--
      ({5.7*.57-.65*\ra*cos(\nextt)},{1.65*\ra*sin(\nextt)})--cycle;
  }
}
\end{scope}

% Hidden rims first; solid strokes are added only on exposed portions.
\tikzset{washer visible rim/.style={solid,line width=.8pt},
         washer hidden rim/.style={densely dashed,line width=.5pt}}
\foreach \q in {.57,.63} {
  \foreach \r in {\q,\q*\q} {
    \draw[figureaccent,washer hidden rim,variable=\t,domain=-90:270,samples=121]
      plot ({5.7*\q-.65*(\r)*cos(\t)},{1.65*(\r)*sin(\t)});
  }
  \draw[figureaccent,washer visible rim,variable=\t,domain=-90:90,samples=81]
    plot ({5.7*\q-.65*\q*cos(\t)},{1.65*\q*sin(\t)});
}
% Only the far inner rim is solid where exposed through the near hole.
\begin{scope}
\clip[variable=\t]
  plot[domain=-90:270,samples=121]
    ({5.7*.63-.65*.63*.63*cos(\t)},{1.65*.63*.63*sin(\t)})--cycle;
\draw[figureaccent,washer visible rim,variable=\t,domain=90:270,samples=81]
  plot ({5.7*.57-.65*.57*.57*cos(\t)},{1.65*.57*.57*sin(\t)});
\end{scope}
% The near outer rear rim is exposed only inside the gray cavity cut.
\begin{scope}
\clip[variable=\u]
  plot[domain=0:1,samples=101] ({5.7*\u},{1.65*\u*\u})
  --plot[domain=1:0,samples=101]
    ({5.7*\u-.65*\u*\u*cos(\washerlowertangent)},
     {1.65*\u*\u*sin(\washerlowertangent)})--cycle;
\draw[figureaccent,washer visible rim,variable=\t,domain=90:270,samples=81]
  plot ({5.7*.63-.65*.63*cos(\t)},{1.65*.63*sin(\t)});
\end{scope}
\draw[figureaccent,washer visible rim,variable=\t,domain=-90:270,samples=121]
  plot ({5.7*.63-.65*.63*.63*cos(\t)},{1.65*.63*.63*sin(\t)});
% The two axial connectors remain solid, with no blue radial cut lines.
\draw[figureaccent,washer visible rim] (3.249,.9405)--(3.591,1.0395);
\draw[figureaccent,washer visible rim,variable=\u,domain=.57:.63,samples=2]
  plot ({5.7*\u-.65*\u*cos(\washerlowertangent)},
        {1.65*\u*sin(\washerlowertangent)});

% Small rim callouts leave the hole and its edges unobstructed.
\draw[figureaccent,line width=.35pt] (3.591,1.0395)--(3.49,1.32);
\node[font=\scriptsize,text=figureaccent,anchor=south] at (3.49,1.32) {$R(x)=x$};
\draw[figureaccent,line width=.35pt]
  ({5.7*.63+.65*.63*.63},0)--(3.96,.15);
\node[font=\scriptsize,text=figureaccent,anchor=west] at (3.98,.15) {$r(x)=x^2$};
\draw[->,dashed,black!45] (-.2,0)--(6.7,0) node[right,text=black] {$x$};
\node[below left] at (0,0) {$0$};
\node[below=3pt] at (5.7,0) {$1$};
\pgfmathsetmacro{\washerleftguide}{.57*(5.7-.65*cos(\washerlowertangent))}
\pgfmathsetmacro{\washerrightguide}{.63*(5.7-.65*cos(\washerlowertangent))}
\foreach \q/\xguide in {.57/\washerleftguide,.63/\washerrightguide} {
  \draw[dashed,black!45] (\xguide,{1.65*\q*sin(\washerlowertangent)})
    --(\xguide,-2.15);
}
\draw[<->] (\washerleftguide,-2.05)--(\washerrightguide,-2.05)
  node[midway,below=3pt] {$dx$};
\end{tikzpicture}
