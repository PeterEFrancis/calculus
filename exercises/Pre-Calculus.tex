\subsection*{A. Prove the following identities.}
\begin{multicols}{3}
\begin{enumerate}
    \item $\tan(x+y)=\frac{\tan(x)+\tan(y)}{1-\tan(x)\tan(y)}$

    \item $\cos(x)\cos(y)=\frac{\cos(x-y)+\cos(x+y)}{2}$
\end{enumerate}

\subsection*{G. Use the properties of logarithms to expand 1--3 and combine 4--6 into a single logarithm. Assume $x,y>0$, and $x>2$ in 3.}
\begin{multicols}{2}
\begin{enumerate}
    \item $\ln(xy^3)$
    \item $\log_2\left(\frac{x^2}{\sqrt y}\right)$
    \item $\ln\left(\frac{(x-2)^2}{x\sqrt{x+1}}\right)$
    \item $2\ln(x)+3\ln(y)$
    \item $\frac12\log_3(x)-2\log_3(y)$
    \item $\ln(x+1)+\ln(x+2)-\ln(x)$
\end{enumerate}
\end{multicols}

\subsection*{H. Solve each equation. Check that your answers make every logarithm defined.}

    \item $\log_3(x)+\log_3(x-2)=1$

\end{multicols}
