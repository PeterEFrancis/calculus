  every node/.style={font=\small,inner sep=2pt}]

% Short segments have slope x+y in these (unequally scaled) axes.
\foreach \a in {-2.25,-1.75,...,3.25}{
  \foreach \b in {.25,.75,...,5.75}{
    \pgfmathsetmacro{\slope}{\a+\b}
    \pgfmathsetmacro{\run}{.1/sqrt(1+(\slope/1.15)^2)}
    \pgfmathsetmacro{\rise}{\slope*\run}
    \draw[gray!35,line width=.35pt]
      ({\a-\run},{\b-\rise})--({\a+\run},{\b+\rise});
  }
}
% Every dashed-curve point appears in the step-size-one table.
\draw[figuresecondary,dashed,line width=.9pt]
  (-1,1)--(0,1)--(1,2)--(2,5)--(3,12);

  \fill[figuresecondary] (\a,\b) circle[radius=1.5pt];

  (-1,1)--(-.5,1)--(0,1.25)--(.5,1.875)--(1,3.0625)
  --(1.5,5.09375)--(2,8.390625)--(2.5,13.5859375);
