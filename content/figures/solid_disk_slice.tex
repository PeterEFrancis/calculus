\begin{tikzpicture}[line width=.7pt,line cap=round,line join=round,
  every node/.style={font=\small,inner sep=2pt}]

\coordinate (conelower) at ({5.3+.5*cos(\coneangle)},{-1.5*sin(\coneangle)});
\fill[black,fill opacity=.08] (0,0)--(coneupper)
  arc[start angle=\coneangle,end angle=-\coneangle,
      x radius=.5,y radius=1.5]--cycle;
\draw[gray!70] (0,0)--(coneupper) (0,0)--(conelower);
\filldraw[fill=black,fill opacity=.04,draw=gray!70]
  (5.3,0) ellipse[x radius=.5,y radius=1.5];
\draw[dashed,gray] (0,0)--(5.3,0);
\fill[figureblue,fill opacity=.12]
  ({5.3*.58},0) ellipse[x radius={.5*.58},y radius={1.5*.58}];
\fill[figureblue,fill opacity=.20]
  ({.58*(5.3+.5*cos(\coneangle))},{.58*1.5*sin(\coneangle)})
  --({.64*(5.3+.5*cos(\coneangle))},{.64*1.5*sin(\coneangle)})
  arc[start angle=\coneangle,end angle=-\coneangle,
      x radius={.5*.64},y radius={1.5*.64}]
  --({.58*(5.3+.5*cos(\coneangle))},{-.58*1.5*sin(\coneangle)})
  arc[start angle=-\coneangle,end angle=\coneangle,
      x radius={.5*.58},y radius={1.5*.58}]--cycle;

  ({5.3*.64},0) ellipse[x radius={.5*.64},y radius={1.5*.64}];

\foreach \q in {.58,.64}{
  \draw[figureaccent,line width=.8pt]
    ({5.3*\q},{1.5*\q})
    arc[start angle=90,end angle=270,x radius={.5*\q},y radius={1.5*\q}];
  \draw[figureaccent,densely dashed,line width=.6pt]
    ({5.3*\q},{-1.5*\q})
    arc[start angle=-90,end angle=90,x radius={.5*\q},y radius={1.5*\q}];

\foreach \sgn in {-1,1}{
  \draw[figureaccent,line width=.8pt]
    ({.58*(5.3+.5*cos(\coneangle))},{\sgn*.58*1.5*sin(\coneangle)})
    --({.64*(5.3+.5*cos(\coneangle))},{\sgn*.64*1.5*sin(\coneangle)});
}
\draw[->,dashed,gray] (-.25,0)--(6.15,0) node[right,text=black] {$x$};
\node[below left] at (0,0) {$0$};

  node[pos=.65,right=12pt] {$f(x)$};

  ({5.3*.58},{-1.5*.58})--({5.3*.58},-1.45)
  ({5.3*.64},{-1.5*.64})--({5.3*.64},-1.45);
\draw[<->] ({5.3*.58},-1.32)--({5.3*.64},-1.32)
  node[midway,below=4pt] {$dx$};
